cmake_minimum_required(VERSION 3.16)
project(runtime_dependency_checks LANGUAGES CXX)
add_executable(runtime_dependency_checks runtime.cpp)
target_compile_features(runtime_dependency_checks PRIVATE cxx_std_17)
target_include_directories(runtime_dependency_checks PRIVATE "${DEPENDENCY_PREFIX}/include")
foreach(lib osg osgDB osgText OpenThreads OpenAL32 avformat avcodec avutil swscale swresample)
    target_link_libraries(runtime_dependency_checks PRIVATE "${DEPENDENCY_PREFIX}/lib/${lib}.lib")
endforeach()
if(MSVC)
    target_compile_options(runtime_dependency_checks PRIVATE /utf-8)
endif()
